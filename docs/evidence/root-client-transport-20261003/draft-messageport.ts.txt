import { ChannelClient, MessagePortProtocol } from '@knorvia/rpc';
import type { IServiceAccessor } from '@knorvia/services';
import { RemoteServiceAccess } from './remoteServiceAccess.js';
import { isRendererProductionBuild } from './rendererLoggingEnv.js';

export interface MessagePortServiceConnection {
  services: IServiceAccessor;
  dispose: (reason?: Error) => void;
}

function connectionDebug(message: string): void {
  if (!isRendererProductionBuild()) console.log(message);
}

export function createMessagePortServiceConnection(port: MessagePort): MessagePortServiceConnection {
  connectionDebug('[messageport] creating protocol and client...');
  const protocol = new MessagePortProtocol(port);
  const client = new ChannelClient(protocol);
  client.onDidInitialize(() => connectionDebug('[messageport] ChannelClient received Initialize from server'));
  connectionDebug('[messageport] client created, waiting for Initialize...');
  const services = new RemoteServiceAccess(client);
  let disposed = false;
  return {
    services,
    dispose(reason) {
      if (disposed) return;
      disposed = true;
      client.dispose(reason);
      protocol.disconnect();
    },
  };
}

export function connectViaMessagePort(port: MessagePort): IServiceAccessor {
  return createMessagePortServiceConnection(port).services;
}
