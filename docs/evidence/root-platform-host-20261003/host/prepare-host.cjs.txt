const bodies=JSON.parse(fs.readFileSync(root+'/host-draft/main-bodies.json','utf8'));
let output=sf.statements.filter(ts.isImportDeclaration).map(n=>n.getText(sf)).join('\n')+'\n';
output+=`import { app, MessageChannelMain, utilityProcess as electronUtilityProcess } from "electron";

for(const n of sf.statements){
 if(ts.isInterfaceDeclaration(n))output+=n.getText(sf)+'\n';
 if(!ts.isFunctionDeclaration(n))continue;
 let signature=n.getText(sf).replace('export declare function','export function').replace(/;$/,'');
 if(n.name.text==='loadWindow')signature=signature.replace('bootstrap?: WindowBootstrapOptions','bootstrap: WindowBootstrapOptions = {}');
 if(n.name.text==='disposeHostProcess')signature=signature.replace('forceKillDelayMs?: number','forceKillDelayMs: number = 300');

 output+=signature+' {\n'+bodies[n.name.text]+'\n}\n';
}
fs.writeFileSync(root+'/host-draft/desktopHostProcess.ts',output);
