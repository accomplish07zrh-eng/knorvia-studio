import {readFile,writeFile} from "node:fs/promises";
import assert from "node:assert/strict";

const results=[],errors=[],failed=[],consoleErrors=[];let context,page;
const capture=async(name)=>{
 await page.screenshot({path:`${p.root}/${name}.png`,fullPage:true});
 const controls=await page.locator('button,input,[role=button],[contenteditable=true],[role=menuitem],[data-testid]').evaluateAll(es=>es.map(e=>({tag:e.tagName,text:e.textContent?.slice(0,180),aria:e.getAttribute('aria-label'),testid:e.getAttribute('data-testid'),placeholder:e.getAttribute('placeholder'),path:e.getAttribute('data-path')})));

try{

 return{legacyShape:fixture,secondaryWorkspace:'beta',legacyRatioAccepted:true,pointerCommitRatio:saved.root.ratio,v2Reload:true,scope:'draft panes only; no synthetic agent snapshots'};
 });

 const legacy={id:'88888888-8888-4888-8888-888888888888',name:'Combined legacy group 中文',goal:'Legacy shared goal',members:['knorvia'],host:'knorvia',sharedSummary:'Legacy shared summary — 中文',mode:'manual',workspaceMode:'isolated',workspacePath:p.workspaces[0],draft:'Legacy unsent group draft — 中文',createdAt:1700000000123,updatedAt:1700000000223};

 });

}finally{await writeFile(p.root+'/web-legacy-results.json',JSON.stringify({results,errors,failed,consoleErrors},null,2));await context?.close();}
