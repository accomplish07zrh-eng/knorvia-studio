setDataBaseDir(p.root+"/data");
const services=createLocalServices({knorviaBuiltinProviderConfigFilePath:p.root+"/builtin-fixture.json",runtimeProcessEnvPatch:{"KNORVIA_STORAGE_DIR": "/tmp/knorvia-ui-combined-902-hjg72jtm/data/.knorvia-studio", "KNORVIA_SESSION_DB_PATH": "/tmp/knorvia-ui-combined-902-hjg72jtm/data/.knorvia-studio/cli/db/db.sqlite"}});
