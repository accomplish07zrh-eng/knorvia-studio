import { basename, dirname, isAbsolute, resolve } from "node:path";
import type { McpServerConfig, RuntimeConfigPatch } from "@knorvia/contracts";
import {
    createWorkspaceHookSourceInput,
    discoverWorkspaceHookConfigPaths,
    workspaceHooksConfigSchema,
    type WorkspaceHookSourceInput,
} from "@knorvia/shared/workspace-hook-discovery";
import { loadFileConfig, type LoadedConfig } from "./file-config.adapter.js";

export interface ProjectConfigFile {
    baseDir: string;
    config: RuntimeConfigPatch;
    diagnostics: LoadedConfig["diagnostics"];
    hookCandidate?: WorkspaceHookSourceInput;
    loaded: boolean;
    path: string;
}

export interface ProjectConfigDiscovery {
    files: ProjectConfigFile[];
    diagnostics: LoadedConfig["diagnostics"];
    hookCandidates: WorkspaceHookSourceInput[];
    loaded: boolean;
    paths: string[];
    mcpServerNames: string[];
}

const CONFIG_DIRECTORY_NAME = ".knorvia-studio";
const STDIO_TRANSPORT = "stdio";
const DEFAULT_SERVER_DIRECTORY = ".";
const FIRST_DISCOVERY_ORDER = 0;
const PENDING_TRUST_CODE = "config_project_hooks_pending_trust";
const PENDING_TRUST_MESSAGE =
    "Project hooks are pending workspace trust and remain blocked";
const HOOKS_CONFIG_PATH = "hooks";
const WARNING_SEVERITY = "warning";

function normalizeServer(
    server: McpServerConfig,
    baseDir: string,
): McpServerConfig {
    if (server.type !== STDIO_TRANSPORT) {
        return server;
    }
    const cwd = server.cwd ?? DEFAULT_SERVER_DIRECTORY;
    return {
        ...server,
        cwd: isAbsolute(cwd) ? cwd : resolve(baseDir, cwd),
    };
}

function normalizeProjectConfig(
    source: RuntimeConfigPatch,
    baseDir: string,
): RuntimeConfigPatch {
    const config = { ...source };
    if (config.hooks) {
        delete config.hooks;
    }
    if (config.mcp?.servers) {
        config.mcp = {
            ...config.mcp,
            servers: Object.fromEntries(
                Object.entries(config.mcp.servers).map(([name, server]) => [
                    name,
                    normalizeServer(server, baseDir),
                ]),
            ),
        };
    }
    return config;
}

export function loadProjectConfigs(
    workingDirectory?: string,
    explicitProjectConfigPath?: string,
): ProjectConfigDiscovery {
    const resolvedDirectory = resolve(workingDirectory ?? process.cwd());
    const discovered = discoverWorkspaceHookConfigPaths({
        workingDirectory: resolvedDirectory,
        ...(explicitProjectConfigPath ? { explicitProjectConfigPath } : {}),
    });
    const files = discovered.map((reference, discoveryOrder) =>
        loadProjectConfigFile(reference.path, {
            discoveryOrder,
            explicitProjectConfig: reference.explicitProjectConfig,
            workingDirectory: resolvedDirectory,
        }),
    );
    return summarizeProjectConfigs(files);
}

export function loadProjectConfigFile(
    path: string,
    options: {
        discoveryOrder?: number;
        explicitProjectConfig?: boolean;
        workingDirectory?: string;
    } = {},
): ProjectConfigFile {
    const result = loadFileConfig(path);
    const containingDirectory = dirname(result.path);
    const baseDir =
        basename(containingDirectory) === CONFIG_DIRECTORY_NAME
            ? dirname(containingDirectory)
            : containingDirectory;
    const diagnostics = [...result.diagnostics];
    const hooks = result.loaded ? result.config.hooks : undefined;
    if (hooks) {
        diagnostics.push({
            code: PENDING_TRUST_CODE,
            filePath: result.path,
            message: PENDING_TRUST_MESSAGE,
            path: HOOKS_CONFIG_PATH,
            severity: WARNING_SEVERITY,
        });
    }
    const config = result.loaded
        ? normalizeProjectConfig(result.config, baseDir)
        : {};
    let hookCandidate: WorkspaceHookSourceInput | undefined;
    if (hooks) {
        const parsedHooks = workspaceHooksConfigSchema.parse(hooks);
        hookCandidate = createWorkspaceHookSourceInput({
            path: result.path,
            workingDirectory: resolve(options.workingDirectory ?? baseDir),
            hooks: parsedHooks,
            discoveryOrder: options.discoveryOrder ?? FIRST_DISCOVERY_ORDER,
            explicitProjectConfig: options.explicitProjectConfig,
        });
    }
    return {
        baseDir,
        config,
        diagnostics,
        ...(hookCandidate ? { hookCandidate } : {}),
        loaded: result.loaded,
        path: result.path,
    };
}

export function summarizeProjectConfigs(
    files: ProjectConfigFile[],
): ProjectConfigDiscovery {
    const diagnostics = files.flatMap((file) => file.diagnostics);
    const loadedFiles = files.filter((file) => file.loaded);
    const hookCandidates: WorkspaceHookSourceInput[] = [];
    const mcpServerNames = new Set<string>();
    for (const file of loadedFiles) {
        if (file.hookCandidate) {
            hookCandidates.push(file.hookCandidate);
        }
        for (const name of Object.keys(file.config.mcp?.servers ?? {})) {
            mcpServerNames.add(name);
        }
    }
    return {
        files: loadedFiles,
        diagnostics,
        hookCandidates,
        loaded: loadedFiles.length > FIRST_DISCOVERY_ORDER,
        paths: loadedFiles.map((file) => file.path),
        mcpServerNames: [...mcpServerNames],
    };
}
