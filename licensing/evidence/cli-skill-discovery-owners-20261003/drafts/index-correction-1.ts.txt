import { realpathSync } from 'node:fs';
import { open, readFile, realpath, stat } from 'node:fs/promises';
import { basename, dirname, join, resolve } from 'node:path';
import type {
  SkillContent,
  SkillDiagnostic,
  SkillLoadOutcome,
  SkillMetadata,
  SkillOperationOptions,
  SkillPort,
  SkillRoot,
} from '@knorvia/contracts';
import { resolveDefaultSkillRoots, type SkillRootResolutionOptions } from './roots.js';
import { scanSkillFilesUnderRoot } from './scan.js';

export interface NodeSkillAdapterOptions extends SkillRootResolutionOptions {
  disabledPaths?: Iterable<string>;
}

const SUPPORTED_FIELDS = ['name', 'description', 'when_to_use', 'license', 'metadata'];
const MANIFEST_PATHS = [
  '.knorvia-plugin/plugin.json',
  '.claude-plugin/plugin.json',
  '.codex-plugin/plugin.json',
  '.cursor-plugin/plugin.json',
];
const DEFAULT_READ_LIMIT = 100000;
const DESCRIPTION_LIMIT = 1024;
const LAST_MANIFEST_DEPTH = 5;
const DELIMITER = '---';

function checkCancellation(options?: SkillOperationOptions): void {
  if (options?.signal?.aborted) throw new Error('Skill operation cancelled');
}

function missingFile(error: unknown): boolean {
  return typeof error === 'object' && error !== null && 'code' in error &&
    (error as { code?: unknown }).code === 'ENOENT';
}

function frontmatterOf(text: string): string | null {
  const normalized = text.replace(/^\uFEFF/, '');
  if (!normalized.startsWith(DELIMITER)) return null;
  const lines = normalized.split(/\r?\n/);
  if (lines[0]?.trim() !== DELIMITER) return null;
  for (let index = 1; index < lines.length; index += 1) {
    if (lines[index]?.trim() === DELIMITER) return lines.slice(1, index).join('\n');
  }
  return null;
}

function bodyOf(text: string): string {
  const normalized = text.replace(/^\uFEFF/, '');
  if (!normalized.startsWith(DELIMITER)) return text;
  const lines = normalized.split(/\r?\n/);
  for (let index = 1; index < lines.length; index += 1) {
    if (lines[index]?.trim() === DELIMITER) return lines.slice(index + 1).join('\n');
  }
  return text;
}

function scalar(value: string | undefined): string | undefined {
  if (value === undefined) return undefined;
  const trimmed = value.trim();
  if (!trimmed) return undefined;
  if ((trimmed.startsWith('"') && trimmed.endsWith('"')) ||
      (trimmed.startsWith("'") && trimmed.endsWith("'"))) {
    return trimmed.slice(1, -1).trim();
  }
  return trimmed;
}

function blockText(lines: string[], folded: boolean): string {
  let indentation: number | undefined;
  for (const line of lines) {
    if (!line.trim()) continue;
    const width = /^\s*/.exec(line)?.[0].length ?? 0;
    indentation = indentation === undefined ? width : Math.min(indentation, width);
  }
  const unindented = lines.map(line => line.trim() ? line.slice(indentation ?? 0) : '');
  if (!folded) return unindented.join('\n').trim();
  const paragraphs: string[] = [];
  let words: string[] = [];
  for (const line of unindented) {
    const value = line.trim();
    if (value) {
      words.push(value);
    } else if (words.length) {
      paragraphs.push(words.join(' '));
      words = [];
    }
  }
  if (words.length) paragraphs.push(words.join(' '));
  return paragraphs.join('\n').trim();
}

function parseFields(header: string, path: string, diagnostics: SkillDiagnostic[]): {
  values: Record<string, string>;
  keys: string[];
} {
  const values: Record<string, string> = {};
  const keys: string[] = [];
  const lines = header.split(/\r?\n/);
  for (let index = 0; index < lines.length; index += 1) {
    const line = lines[index] ?? '';
    const trimmed = line.trim();
    if (!trimmed || trimmed.startsWith('#') || /^\s/.test(line)) continue;
    const colon = line.indexOf(':');
    if (colon <= 0) {
      diagnostics.push({
        code: 'skill_invalid_frontmatter',
        severity: 'warning',
        message: 'Invalid frontmatter line ' + (index + 1) + ' in ' + basename(path),
        path,
      });
      continue;
    }
    const key = line.slice(0, colon).trim();
    keys.push(key);
    const value = line.slice(colon + 1).trim();
    const folded = /^>[+-]?$/.test(value);
    if (folded || /^\|[+-]?$/.test(value)) {
      const block: string[] = [];
      while (index + 1 < lines.length) {
        const next = lines[index + 1] ?? '';
        if (next.trim() && !/^\s/.test(next)) break;
        block.push(next);
        index += 1;
      }
      values[key] = blockText(block, folded);
    } else {
      values[key] = value;
    }
  }
  return { values, keys };
}

async function findPluginName(resolvedRoot: string): Promise<string | undefined> {
  let current = resolve(resolvedRoot);
  for (let depth = 0; depth <= LAST_MANIFEST_DEPTH; depth += 1) {
    for (const relative of MANIFEST_PATHS) {
      const manifestPath = join(current, relative);
      try {
        const manifest = JSON.parse(await readFile(manifestPath, 'utf8'));
        if (typeof manifest.name === 'string' && manifest.name.trim().length > 0) {
          return manifest.name.trim();
        }
      } catch {
        // Manifest misses and malformed values do not affect skill admission.
      }
    }
    const parent = dirname(current);
    if (parent === current) break;
    current = parent;
  }
  return undefined;
}

export class NodeSkillAdapter implements SkillPort {
  private readonly factoryOptions: NodeSkillAdapterOptions;
  private readonly disabled = new Set<string>();
  private readonly pluginNames = new Map<string, Promise<string | undefined>>();

  constructor(options: NodeSkillAdapterOptions = {}) {
    this.factoryOptions = options;
    const paths = Array.from(options.disabledPaths ?? []);
    for (const path of paths) {
      const resolved = resolve(path);
      let canonical: string | undefined;
      try {
        canonical = realpathSync(resolved);
      } catch {
        canonical = undefined;
      }
      this.disabled.add(resolved);
      if (canonical && canonical !== resolved) this.disabled.add(canonical);
    }
  }

  private async labels(root: SkillRoot, skillName: string): Promise<{
    pluginId?: string;
    pluginName?: string;
    qualifiedName?: string;
  }> {
    if (root.source !== 'plugin') return {};
    const resolvedRoot = resolve(root.path);
    let pending = this.pluginNames.get(resolvedRoot);
    if (pending === undefined) {
      pending = findPluginName(resolvedRoot);
      this.pluginNames.set(resolvedRoot, pending);
    }
    const pluginName = await pending;
    if (!pluginName) return {};
    return {
      ...(root.pluginId ? { pluginId: root.pluginId } : {}),
      pluginName,
      qualifiedName: pluginName + ':' + skillName,
    };
  }

  private async parse(path: string, root: SkillRoot, diagnostics: SkillDiagnostic[]): Promise<SkillMetadata | null> {
    let text: string;
    try {
      text = await readFile(path, 'utf8');
    } catch (error) {
      if (!missingFile(error)) diagnostics.push({
        code: 'skill_read_failed',
        severity: 'warning',
        message: error instanceof Error ? error.message : 'Failed to read skill: ' + path,
        path,
      });
      return null;
    }
    const header = frontmatterOf(text);
    const { values, keys } = header ? parseFields(header, path, diagnostics) : { values: {}, keys: [] } as {
      values: Record<string, string>;
      keys: string[];
    };
    const name = scalar(values.name) ?? (header ? undefined : basename(dirname(path)));
    if (!name) {
      diagnostics.push({
        code: 'skill_missing_name',
        severity: 'error',
        message: 'Skill frontmatter must include a name: ' + path,
        path,
      });
      return null;
    }
    const rawDescription = scalar(values.description);
    if (!rawDescription && header) {
      diagnostics.push({
        code: 'skill_missing_description',
        severity: 'error',
        message: 'Skill frontmatter must include a description: ' + path,
        path,
        skillName: name,
      });
      return null;
    }
    const description = rawDescription ?? '';
    if (description.length > DESCRIPTION_LIMIT) {
      diagnostics.push({
        code: 'skill_description_too_long',
        severity: 'error',
        message: 'Skill description is too long: ' + name,
        path,
        skillName: name,
      });
      return null;
    }
    const alias = await this.labels(root, name);
    return {
      name,
      description,
      whenToUse: scalar(values.when_to_use),
      ...alias,
      path,
      directory: dirname(path),
      rootPath: root.path,
      scope: root.scope,
      source: root.source,
      safeToAutoLoad: keys.every(key => SUPPORTED_FIELDS.includes(key)),
      frontmatterKeys: keys,
      policy: { allowImplicitInvocation: true },
    };
  }

  async discoverSkills(request: { workingDirectory: string; roots?: SkillRoot[] }, options?: SkillOperationOptions): Promise<SkillLoadOutcome> {
    checkCancellation(options);
    const cwd = resolve(request.workingDirectory);
    const roots = request.roots ?? await resolveDefaultSkillRoots(cwd, this.factoryOptions);
    const orderedRoots = roots.toSorted((left, right) => left.priority - right.priority);
    const diagnostics: SkillDiagnostic[] = [];
    const selected = new Map<string, SkillMetadata>();
    let totalDiscovered = 0;
    for (const root of orderedRoots) {
      checkCancellation(options);
      let paths: string[];
      try {
        paths = await scanSkillFilesUnderRoot(root.path, { followSymbolicLinks: root.source !== 'plugin' });
      } catch (error) {
        if (!missingFile(error)) diagnostics.push({
          code: 'skill_scan_failed',
          severity: 'warning',
          message: error instanceof Error ? error.message : 'Failed to scan skill root: ' + root.path,
          path: root.path,
        });
        paths = [];
      }
      for (const path of paths) {
        checkCancellation(options);
        const metadata = await this.parse(path, root, diagnostics);
        if (!metadata) continue;
        const resolved = resolve(metadata.path);
        if (this.disabled.has(resolved)) continue;
        const canonical = await realpath(resolved).catch(() => resolved);
        if (this.disabled.has(canonical)) continue;
        totalDiscovered += 1;
        if (!selected.has(metadata.path)) selected.set(metadata.path, metadata);
      }
    }
    const skills = Array.from(selected.values()).toSorted((left, right) => left.name.localeCompare(right.name));
    return { skills, diagnostics, totalDiscovered };
  }

  async loadSkill(request: { name: string; workingDirectory: string; roots?: SkillRoot[]; maxBytes?: number }, options?: SkillOperationOptions): Promise<SkillContent> {
    checkCancellation(options);
    const discovered = await this.discoverSkills({ workingDirectory: request.workingDirectory, roots: request.roots }, options);
    const metadata = discovered.skills.find(skill => skill.name === request.name || skill.qualifiedName === request.name);
    if (!metadata) throw new Error('Skill not found: ' + request.name);
    const maxBytes = request.maxBytes ?? DEFAULT_READ_LIMIT;
    const info = await stat(metadata.path);
    const truncated = info.size > maxBytes;
    let buffer: Buffer;
    if (truncated) {
      const handle = await open(metadata.path, 'r');
      try {
        const allocated = Buffer.alloc(Math.max(0, maxBytes));
        const { bytesRead } = await handle.read(allocated, 0, allocated.byteLength, 0);
        buffer = allocated.subarray(0, bytesRead);
      } finally {
        await handle.close();
      }
    } else {
      buffer = await readFile(metadata.path);
    }
    return {
      metadata,
      content: bodyOf(buffer.toString('utf8')).trim(),
      baseDirectory: metadata.directory,
      bytesRead: buffer.byteLength,
      sizeBytes: info.size,
      truncated,
    };
  }
}

export function createNodeSkillAdapter(options: NodeSkillAdapterOptions = {}): NodeSkillAdapter {
  return new NodeSkillAdapter(options);
}
